void handleRoot() {
  if (server.hasArg("FND")) {
    String fndVal = server.arg("FND");
    int fndvalue = fndVal.toInt(); // 0 ~ 255 사이의 정수 변환
    aaa.setDisplayToDecNumber(fndvalue,0);
    
    
    Serial.print("현재 FND 값: ");
    Serial.println(fndvalue);

    
    aaa.setLEDs(fndvalue); 
    
    server.send(200, "text/plain", "OK");
    return;
  }
  
  // 처음 접속할 때만 전체 HTML 페이지 전송
  server.send(200, "text/html", MAIN_page);
}

void setup() {
  Serial.begin(9600);
  WiFi.softAP(ssid, password);
  Serial.println();
  Serial.print("AP IP Address: ");
  Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot); 
  server.begin();             
  
  // [체크포인트] 부팅 직후 모듈 정상 작동 테스트를 위해 LED 8개를 모두 켭니다.
  aaa.setLEDs(255); 
}

void loop() {
  server.handleClient(); 
}
